# 3. Execute the proposed restoration selection rule in a real testServer.
shiny::testServer(function(input, output, session) {
  applied_params <- shiny::reactiveVal(list(impute_method = "min"))
  output$ui_impute <- shiny::renderUI({
    offered <- c("min", "knn", "MinProb")
    sel <- shiny::isolate(input$impute_method)
    if (is.null(sel) || !sel %in% offered) {
      sel <- applied_params()$impute_method %||% "MinProb"
    }
    if (!sel %in% offered) sel <- "none"
    shiny::selectInput("impute_method", "Imputation", choices = offered, selected = sel)
  })
}, {
  session$setInputs(impute_method = "knn")
  applied_params(list(impute_method = "min"))
  rendered <- output$ui_impute$html
  emit("T2_1_RESTORED_MIN_BUT_KNN_SELECTED", grepl('value="knn" selected', rendered, fixed = TRUE))
})
