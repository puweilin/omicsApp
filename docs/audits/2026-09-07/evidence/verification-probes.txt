 session$setInputs(`diff-method`='ttest',`diff-group_col`='group',`diff-control`='A',`diff-case`='B',`diff-rerun`=1)

 cat('PROJECT_B_AFTER_OPEN_CONTRAST=',current_project()$bundles$diff$params$comparison,'; TITLE=',current_project()$name,'\n',sep='')
})
# The integration result loses both input effects, and the plotting fallback sets B to zero.
p<-omicsCore::omics_project('two',experiments=list(a=i,b=i))
it<-omicsCore::run_integration(p,method='concordance',diff_bundles=list(a=bA,b=bB))
pdf<-omicsCore::plot_integration(it,view='effect_pair')$data
cat('TRUE_B_EFFECT_NONZERO=',sum(bB$results$diff_result_df$effect!=0),'; PLOTTED_B_NONZERO=',sum(pdf$effect_b!=0),'; PLOTTED_A_EQUALS_DIFFERENCE=',all(pdf$effect_a==pdf$effect),'\n',sep='')
# t-test accepts covariates but removes them without a warning or recorded exclusion.
warn<-character();bc<-withCallingHandlers(omicsCore::run_diff(i,method='ttest',group_col='group',control_group='A',case_group='B',covariates='age'),warning=function(w){warn<<-c(warn,conditionMessage(w));invokeRestart('muffleWarning')})
cat('TTEST_COVARIATE_RECORDED=',!is.null(bc$params$covariates),'; RESULT_EQUALS_UNADJUSTED=',identical(bc$results$diff_result_df,bA$results$diff_result_df),'; WARNINGS=',length(warn),'\n',sep='')
# Loading NULL upstream must clear an enrichment, but observeEvent ignores it.
db<-shiny::reactiveVal(bA)
shiny::testServer(omicsApp:::enrich_view_server,args=list(diff_bundle=db),{
 session$flushReact();enrich_bundle(omicsCore::new_analysis_bundle('run_enrichment'));is_demo(FALSE)

})
