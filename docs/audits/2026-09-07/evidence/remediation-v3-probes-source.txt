nf <- 30L; ns <- 20L
mat <- matrix(rnorm(nf * ns, 20, 1), nf, ns,
              dimnames = list(paste0("f", 1:nf), paste0("s", 1:ns)))
meta <- data.frame(group = factor(rep(c("A", "v2"), each = ns / 2)),
                   age = rnorm(ns), row.names = colnames(mat))
fd <- data.frame(feature_id = rownames(mat), feature_symbol = paste0("G", 1:nf),
                 row.names = rownames(mat))

# Positive regression: v3's safe coefficient mapping fixes v2's collision.
mm <- candidate$build_design(meta, c("group", "age"), primary = "group", intercept = FALSE)
ct <- candidate$design_contrast(mm, "A", "v2")
ref_mm <- cbind(A = as.numeric(meta$group == "A"), B = as.numeric(meta$group == "v2"), age = meta$age)
ref_ct <- matrix(c(-1, 1, 0), 3, 1)
fit <- limma::eBayes(limma::contrasts.fit(limma::lmFit(mat, mm), ct))
ref <- limma::eBayes(limma::contrasts.fit(limma::lmFit(mat, ref_mm), ref_ct))
emit("V3_CONTRAST_WEIGHTS", as.numeric(ct))
emit("V3_EFFECT_MAXDIFF", max(abs(fit$coefficients - ref$coefficients)))
emit("V3_P_MAXDIFF", max(abs(fit$p.value - ref$p.value)))
meta_many <- meta
for (i in 1:9) meta_many[[paste0("cov", i)]] <- rnorm(ns)
mm_many <- candidate$build_design(meta_many, names(meta_many), primary = "group", intercept = FALSE)
emit("V3_MANY_TERMS_CONTRAST", outcome(candidate$design_contrast(mm_many, "A", "v2")))
marker <- tempfile()
meta_payload <- meta
names(meta_payload)[1L] <- sprintf('file.create("%s")', marker)
mm_payload <- candidate$build_design(meta_payload, names(meta_payload), primary = names(meta_payload)[1L], intercept = FALSE)
fit_payload <- limma::eBayes(limma::contrasts.fit(limma::lmFit(mat, mm_payload),
                              candidate$design_contrast(mm_payload, "A", "v2")))
emit("V3_PAYLOAD_MARKER_CREATED", file.exists(marker))
emit("V3_PAYLOAD_EFFECT_MAXDIFF", max(abs(fit_payload$coefficients - ref$coefficients)))

# Capture GSEA's effective output cutoff through the real bundle adapter.
probe_gsea_cutoff <- function() {
  seen_cutoff <- NULL
  testthat::local_mocked_bindings(run_gsea_database = function(..., p_cutoff) { seen_cutoff <<- p_cutoff; NULL }, .package = "omicsCore")
  get("run_gsea_from_bundle", asNamespace("omicsCore"))(raw_diff, database = "hallmark")
  emit("CURRENT_GSEA_OUTPUT_CUTOFF_DEFAULT", seen_cutoff)
  get("run_gsea_from_bundle", asNamespace("omicsCore"))(raw_diff, database = "hallmark", output_p_cutoff = 1)
  emit("GSEA_OUTPUT_CUTOFF_EXPLICIT_COMPLETE", seen_cutoff)
}
probe_gsea_cutoff()
